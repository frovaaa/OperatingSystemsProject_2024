\subsection*{/pintos/userprog/syscall.c}

\begin{itemize}
    \item (\textit{added}) \textbf{imports}
    
    \item (\textit{added}) method \textbf{signatures} for newly implemented methods
    
    \item (\textit{added}) \textbf{struct item \{int fd; struct file *file; struct hash\_elem elem;\}}: Needed to create hash table to map ints (fds) to file pointers. We decided to go with the globa table shared between threads (with a lock to access it)
    
    \item (\textit{added}) \textbf{\textit{method}} \textbf{unsigned item\_hash(const struct hash\_elem *e, void *aux)}: required for hashtable definition and usage
    
    \item (\textit{added}) \textbf{\textit{method}} \textbf{bool item\_compare(const struct hash\_elem *a, const struct hash\_elem *b, void *aux)}: required for hashtable definition and usage
    
    \item (\textit{added}) \textbf{\textit{gloabal variable}} \textbf{struct hash file\_table}: the actual hash table 
    
    \item (\textit{added}) \textbf{\textit{gloabal variable}} \textbf{struct semaphore file\_table\_lock}: the lock to protect the table 
    
        \item (\textit{added}) \textbf{\textit{gloabal variable}} \textbf{unsigned next\_fd}: a number to keep track of the unique hash table entries (files)
        
    \item (\textit{modified}) \textbf{void syscall\_init (void)}: 
    \begin{itemize}
        \item[] Added initialization of the hashtable and the lock
        \item[] Added the syscall method references in the call[]
    \end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_halt (struct intr\_frame *f)}: Calling \textbf{shutdown\_power\_of()}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_create (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file name from the stack
    \item[] Check if the pointer to the filename is valid using provided method
    \item[] Get the initial size of the file from the stack
    \item[] Create the file and store the result of this operation in a local variable
    \item[] Put the result of the creation of the file in f->eax 
    \item[] Release the lock
\end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_remove (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file name from the stack
    \item[] Check if the pointer to the filename is valid using provided method
    \item[] Remove the file and store the result of this operation in a local variable
    \item[] Put the result of the operation in f->eax 
    \item[] Release the lock
\end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_filesize (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file descriptor (fd) from the stack
    \item[] Get the corresponding file from the hashtable
    \item[] Check if what was retrieved is null, in that case release the lock and return the error code on f->eax
    \item[] Otherwise get the file pointer from the hash table entry that we got
    \item[] Get the filesize using the provided method file\_length
    \item[] Put the result of the operation in f->eax 
    \item[] Release the lock
\end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_close (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file descriptor (fd) from the stack
    \item[] Get the corresponding file from the hashtable
    \item[] Check if what was retrieved is null, in that case release the lock
    \item[] Otherwise get the file pointer from the hash table entry that we got
    \item[] Close the file using the provided method file\_close
    \item[] Remove the entry from the hash table
    \item[] Release the lock
\end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_tell (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file descriptor (fd) from the stack
    \item[] Get the corresponding file from the hashtable
    \item[] Check if what was retrieved is null, in that case release the lock and return the error code on f->eax
    \item[] Otherwise get the file pointer from the hash table entry that we got
    \item[] Get the file position using the provided method file\_tell
    \item[] Put the result of the operation in f->eax 
    \item[] Release the lock
\end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_seek (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file descriptor (fd) from the stack
    \item[] Get the file position from the stack
    \item[] Get the corresponding file from the hashtable
    \item[] Check if what was retrieved is null, in that case release the lock 
    \item[] Otherwise get the file pointer from the hash table entry that we got
    \item[] Seek the file using the provided method file\_seek
    \item[] Release the lock
\end{itemize}

    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_read (struct intr\_frame *f)}: 
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file descriptor (fd) from the stack
    \item[] Get the buffer pointer from the stack
    \item[] Get the buffer size from the stack
    \item[] Check the validity of the buffer pointer, if it's invalid release the lock and return the error code on f->eax
    \item[] Check if the file descriptor is 0 (stdin), if yes then we are reading from the keyboard so we fill the buffer using input\_getc(), then we push the size on f->eax and we release the lock.
        \item[] Otherwise, get the corresponding file from the hashtable
    \item[] Check if what was retrieved is null, in that case release the lock and return the error code on f->eax
    \item[] Otherwise get the file pointer from the hash table entry that we got
    \item[] Read from the file and save the result in the buffer using the provided method file\_read
    \item[] Put how many bytes we read in f->eax
    \item[] Release the lock
\end{itemize}


    \item (\textit{added}) \textbf{\textit{method}} \textbf{static void
syscall\_open (struct intr\_frame *f)}:
\begin{itemize}
    \item[] Acquire the lock to access the file table 
    \item[] Get the file pointer from the stack
    \item[] Check the validity of the file pointer, if it's invalid release the lock and return the error code on f->eax
    \item[] Otherwise, open the file with provided method filesys\_open and save its pointer in file\_opened
    \item[] Check if the file was opened, if not then release the lock and return the error code on f->eax
    \item[] Otherwise, create a new struct item *file\_item in order to put the file in the hash table
    \item[] Set the fd as the next\_ft and the file pointer to the file\_opened
    \item[] Instert the newly created element in the hashtable
    \item[] Put the next\_fd in f->eax (because we return the fd of the newly opened file)
    \item[] Increment the global next\_fd
    \item[] Release the lock
\end{itemize}
    
\end{itemize}
